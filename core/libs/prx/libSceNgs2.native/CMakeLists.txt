include(${CMAKE_CURRENT_SOURCE_DIR}/Ngs2Library.cmake)
add_sce_ngs2_library(libSceNgs2.native)

if(BUILD_TESTING)
    foreach(ngs2Test IN ITEMS Render Atrac9 Filter Custom Parse Stream UserFx Report Geom)
        string(TOLOWER ${ngs2Test} testName)
        add_test_executable(ngs2_${testName}_tests ${CMAKE_CURRENT_SOURCE_DIR}/tests/Ngs2${ngs2Test}.cpp)
        target_include_directories(ngs2_${testName}_tests PRIVATE ${LIBS_INCLUDE_DIR} ${CMAKE_SOURCE_DIR}/3rdparty/LibAtrac9/C/src)
        target_link_libraries(ngs2_${testName}_tests PRIVATE libSceNgs2.native atrac9 libkernel libc)
        configure_windows_unwind(ngs2_${testName}_tests)
        add_test(NAME ngs2_${testName} COMMAND ngs2_${testName}_tests)
        if(WIN32)
            set_tests_properties(ngs2_${testName} PROPERTIES ENVIRONMENT_MODIFICATION
                "PATH=path_list_prepend:$<TARGET_FILE_DIR:libSceNgs2.native>")
        endif()
    endforeach()
    set_tests_properties(ngs2_geom PROPERTIES TIMEOUT 30 LABELS "module-functional;module:libSceNgs2.native")
    add_test_executable(ngs2_core_geom_tests ${CMAKE_CURRENT_SOURCE_DIR}/tests/Ngs2Geom.cpp)
    target_include_directories(ngs2_core_geom_tests PRIVATE ${LIBS_INCLUDE_DIR} ${CMAKE_SOURCE_DIR}/3rdparty/LibAtrac9/C/src)
    target_link_libraries(ngs2_core_geom_tests PRIVATE libSceNgs2 atrac9 libkernel libc)
    configure_windows_unwind(ngs2_core_geom_tests)
    add_test(NAME ngs2_core_geom COMMAND ngs2_core_geom_tests)
    set_tests_properties(ngs2_core_geom PROPERTIES TIMEOUT 30 LABELS "module-functional;module:libSceNgs2.native")
    if(WIN32)
        set_tests_properties(ngs2_core_geom PROPERTIES ENVIRONMENT_MODIFICATION
            "PATH=path_list_prepend:$<TARGET_FILE_DIR:libSceNgs2>")
    endif()
endif()
