project(AnyPS5DenormalProofTests LANGUAGES CXX)
enable_testing()
add_executable(denormal_value_proof_tests DenormalValueProofTests.cpp)
target_include_directories(denormal_value_proof_tests PRIVATE ../../Optimization/include)
