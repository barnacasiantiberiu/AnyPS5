cmake_minimum_required(VERSION 3.20)
project(AnyPS5ShaderResourceTests LANGUAGES CXX)
enable_testing()
find_package(Threads REQUIRED)
option(SHADER_RESOURCE_SANITIZERS "Enable address and undefined behavior sanitizers" OFF)
get_filename_component(repoRoot "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
foreach(suite IN ITEMS Cache BdaRange)
    add_executable(${suite}Tests ${suite}Tests.cpp)
    target_compile_features(${suite}Tests PRIVATE cxx_std_20)
    target_include_directories(${suite}Tests PRIVATE "${repoRoot}/core/shader/recompiler")
    target_link_libraries(${suite}Tests PRIVATE Threads::Threads)
    set_target_properties(${suite}Tests PROPERTIES CXX_EXTENSIONS OFF)
    if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
        target_compile_options(${suite}Tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
        if(SHADER_RESOURCE_SANITIZERS)
            target_compile_options(${suite}Tests PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
            target_link_options(${suite}Tests PRIVATE -fsanitize=address,undefined)
        endif()
    elseif(SHADER_RESOURCE_SANITIZERS)
        message(FATAL_ERROR "The requested sanitizers require GCC or Clang")
    endif()
    add_test(NAME shader_resource_${suite} COMMAND ${suite}Tests)
    set_tests_properties(shader_resource_${suite} PROPERTIES TIMEOUT 30)
endforeach()
